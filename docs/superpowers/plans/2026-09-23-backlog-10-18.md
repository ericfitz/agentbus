### Task 3: #13 tag matching by indexed join

**Human decision (2026-09-23, user):** store tag subscriptions so that matching messages come from a join on indexed fields, with no per-message iteration. Record this decision in ADR 0009 as an amendment labeled "Human decision".

**Design:**
- New table `tag_subscription_tags(sender, tags_key, tag)` holds one row per tag of each AND set. Its primary key is `(sender, tags_key, tag)`, with a foreign key `(sender, tags_key)` → `tag_subscriptions` `ON DELETE CASCADE`. Every existing `DELETE FROM tag_subscriptions` (receive.go:314, sessions.go:165,180, UnsubscribeTags) therefore cleans up without code changes.
- A new index `message_tags_tag_seq ON message_tags(tag, seq)` replaces `message_tags_tag(tag)`. `message_tags` is a rowid table, so the old index can't range-scan by seq.
- `tagCond` drives the match from the sender's few subscribed tags into `message_tags` by `(tag, seq > floor)`, so it reads only tagged messages above the cursor. Untagged traffic is never visited. A message matches a set when its count of matching tag rows equals the set's size.
- Schema v5: migration step `4` creates the table (the DDL does that), backfills it from the existing `tags_key` values, and drops the old index. 1.7.0 ships schema v5, and the dev bus migrates v4→v5 on its next open.

**Files:**
- Modify: `internal/bus/schema.go` (the table, the index, `schemaVersion = 5`), `internal/bus/migrate.go` (step 4), `internal/bus/tags.go` (`SubscribeTags` inserts the per-tag rows, `tagCond` signature and SQL), `internal/bus/receive.go` (~line 470, `srcCond`), `internal/bus/wait.go` (~line 95), `internal/bus/admin.go:118` (add the table to the list, before `tag_subscriptions`), `docs/adr/0009-message-tags-and-tag-subscriptions.md`

func splitTagSets(tx *sql.Tx) error {
	if _, err := tx.Exec(`CREATE TABLE IF NOT EXISTS tag_subscription_tags (...same DDL...)`); err != nil {
		return err
	}
	if _, err := tx.Exec(`WITH RECURSIVE split(sender, tags_key, tag, rest) AS (
		SELECT sender, tags_key, '', tags_key || ',' FROM tag_subscriptions
		UNION ALL
		SELECT sender, tags_key, substr(rest, 1, instr(rest, ',') - 1), substr(rest, instr(rest, ',') + 1) FROM split WHERE rest <> '')

		return err
	}
	_, err := tx.Exec("DROP INDEX IF EXISTS message_tags_tag")
	return err
}

```go
// tagCond is the tag source's predicate over messages: ordinary chat
// channels only (dm/ inboxes excluded by name; memory channels and task
// lists are memory-kind), skipping channels the sender is directly
// subscribed to, and carrying every tag of at least one set added before
// the message. Matching is a join from the sender's set tags into
// message_tags(tag, seq) above floor, so only tagged messages are read
// (#13); a message matches a set when it has as many of the set's tags as
// the set has.
func tagCond(as string, floor int64) (string, []any) {
	return `messages.seq IN (
		SELECT mt.seq FROM tag_subscription_tags st
		JOIN tag_subscriptions ts ON ts.sender=st.sender AND ts.tags_key=st.tags_key
		JOIN message_tags mt ON mt.tag=st.tag AND mt.seq>max(ts.created_seq, ?)
		WHERE st.sender=?
		GROUP BY mt.seq, st.tags_key
		HAVING count(*)=(SELECT count(*) FROM tag_subscription_tags c WHERE c.sender=st.sender AND c.tags_key=st.tags_key))
	AND messages.channel IN (SELECT name FROM channels WHERE kind='ordinary' AND name NOT LIKE 'dm/%')
	AND messages.channel NOT IN (SELECT channel FROM subscriptions WHERE sender=?)`, []any{floor, as, as}
}
```

Update the callers: in receive.go `srcCond`, `tagCond(as, s.cursor)`; in wait.go, `tagCond(as, floor)`. Wait still needs `sets` only to skip an empty tag source, so keep the `len(sets) == 0` check. If the plan test shows `SCAN` on `mt`, `max(...)` is blocking the range: split it into `mt.seq>? AND mt.seq>ts.created_seq`.

// emojiIcons is the default set, captured from the variables' initial values.
var emojiIcons = func() map[string]string {
	s := map[string]string{}
	for k, p := range iconVars {
		s[k] = *p
	}
	return s
}()

// nerdIcons is the user's preferred Font Awesome iconography (issue #17
// attachment agentbus-tui-icon-map.md) at the codepoints Nerd Fonts 3.x
// gives it. Nerd Fonts carries Font Awesome Free in its "regular" (FA4 _o)
// or solid form; the comment names the Nerd Font glyph where it differs
// from the requested FA6 name.
var nerdIcons = map[string]string{
	"chat":             "\uf27a", // fa-message
	"memory":           "\uf0c7", // fa-floppy-disk (nf fa-save)
	"tasks":            "\uf0ae", // fa-list-check (nf fa-tasks, its FA5 name)
	"agent":            "\uee0d", // fa-robot
	"user":             "\uf007", // fa-user (solid; no regular user in Nerd Fonts)
	"idle":             "\U000f04b2", // md-sleep "zZ"; fa-face-sleeping is Pro-only
	"task_pending":     "\uf096", // fa-square regular (nf fa-square_o)
	"task_in_progress": "\uf152", // fa-square-caret-right regular (nf fa-toggle_right)
	"task_completed":   "\uf046", // fa-square-check regular (nf fa-check_square_o)
	"collapsed":        "\uf0da", // fa-caret-right
	"expanded":         "\uf0d7", // fa-caret-down
	"error":            "\uf06a", // fa-circle-exclamation (nf fa-exclamation_circle)
	"arrow":            "\uf061", // fa-arrow-right
}

// padIcon pads a glyph to its emoji counterpart's layout: the wide icons
// take three columns (two for the emoji, one space) so rails and task rows
// line up whichever set is active; the arrow gets a space on each side, the
// error prefix one trailing space, and the thread marks stay bare.
func padIcon(name, g string) string {
	switch name {
	case "arrow":
		return " " + g + " "
	case "error":
		return g + " "
	case "collapsed", "expanded":
		return g
	}
	return g + strings.Repeat(" ", max(3-lipgloss.Width(g), 1))
}

func setIcons(set map[string]string) {
	for k, p := range iconVars {
		*p = set[k]
	}
}

// LoadIcons applies cfg.Icons: emoji (default), nerdfont, or custom, where
// icon_map overrides any subset of emoji. An unknown set, an unknown name, or
// an empty value warns one line and falls back to emoji, like LoadTheme.
func LoadIcons(cfg config.Config, warn io.Writer) string {
	set, name := emojiIcons, strings.ToLower(strings.TrimSpace(cfg.Icons))
	switch name {
	case "", "emoji":
		name = "emoji"
	case "nerdfont":
		set = map[string]string{}
		for k, g := range nerdIcons {
			set[k] = padIcon(k, g)
		}
	case "custom":
		set = maps.Clone(emojiIcons)
		for k, g := range cfg.IconMap {
			if _, ok := emojiIcons[k]; !ok || g == "" {
				_, _ = fmt.Fprintf(warn, "agentbus tui: icon_map %q=%q is not a known icon with a value; using emoji\n", k, g)
				continue
			}
			set[k] = padIcon(k, g)
		}
	default:
		_, _ = fmt.Fprintf(warn, "agentbus tui: icons %q is not emoji, nerdfont, or custom; using emoji\n", cfg.Icons)
		name = "emoji"
	}
	setIcons(set)
	return name
}
